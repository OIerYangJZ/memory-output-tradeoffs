The lower bounds rest on how far apart the target unitaries of a layer
are, so we first compute that separation and fix the grid.  For an
alphabet $[K]_0$ and a step $\Delta>0$ put
$\operatorname{dist}_{2\pi}(a,0):=\min_{z\in\mathbb Z}|a-2\pi z|$,
\begin{gather*}
 \gamma(K,\Delta):=\min_{1\leq d\leq K-1}
       \operatorname{dist}_{2\pi}(d\Delta,0),\\
 U_x:=\prod_{j=1}^mR_{P_j}(\Delta x_j),
\end{gather*}
for $x\in[K]_0^m$ and independent characters $P_1,\ldots,P_m$.

\begin{theorem}[Exact worst-pair separation]
\label{thm:w1-exact-separation}
For every $m\geq1$,
\begin{equation}
 \min_{x\neq y}d_{\mathrm{proj}}(U_x,U_y)
 =2\sin\!\left(\frac{\gamma(K,\Delta)}4\right).
 \label{eq:w1-exact-separation}
\end{equation}
If $\Delta=2\pi/Q$ and $2\leq K\leq Q$, then $\gamma(K,\Delta)=\Delta$:
the full residue alphabet $K=Q$ is allowed although the coefficient
interval $[0,(K-1)\Delta]$ wraps around the circle.
\end{theorem}

\begin{proof}
Fix $x\neq y$ and $j$ with $s_j:=x_j-y_j\neq0$.  Independence makes the
joint sign map $z\mapsto(\chi_{P_1}(z),\ldots,\chi_{P_m}(z))$ surjective
onto $\{\pm1\}^m$, so the diagonal unitary $U_xU_y^\dagger$ has two
eigenvalues, from sign patterns that agree off $j$, at circular angular
separation $\operatorname{dist}_{2\pi}(|s_j|\Delta,0)\geq\gamma$.  After
any common phase rotation, one of two circle points at angular separation
$g\in[0,\pi]$ lies at chordal distance at least $2\sin(g/4)$ from $1$;
taking the infimum over the global phase gives the lower bound.  A pair
that differs in one coordinate by a minimizing $d$ has exactly two
relative eigenvalues at separation $\gamma$, and centering the shorter arc
attains the bound.  For $\Delta=2\pi/Q$ and $1\leq d\leq Q-1$,
$\operatorname{dist}_{2\pi}(d\Delta,0)=(2\pi/Q)\min\{d,Q-d\}\geq2\pi/Q$,
with equality at $d=1$.
\end{proof}

\begin{corollary}[The $\epsilon$-calibrated grid]
\label{cor:w1-epsilon-cyclic}
\label{lem:grid-calibration}
For $0<\epsilon\leq1/8$ put
\begin{equation}
 Q_\epsilon:=\left\lfloor
 \frac{\pi}{2\arcsin(2\epsilon)}\right\rfloor,
 \qquad K_\epsilon:=Q_\epsilon,
 \qquad \Delta_\epsilon:=\frac{2\pi}{Q_\epsilon}.
 \label{eq:w1-epsilon-grid}
\end{equation}
Then $1/(4\epsilon)\leq K_\epsilon\leq\pi/(4\epsilon)$; the
$K_\epsilon^m$ unitaries $\{U_x:x\in\mathbb Z_{Q_\epsilon}^m\}$ are
pairwise at projective distance at least
$2\sin(\pi/(2Q_\epsilon))\geq4\epsilon$; and $Q_\epsilon$ equals the
largest number of rotations $R_P(\theta)$ about one axis that are
pairwise $4\epsilon$-separated, i.e., at projective distance at least
$4\epsilon$.  At $\epsilon=10^{-3}$,
$K_\epsilon=785$.
\end{corollary}

The proof is in Appendix~\ref{app:proofs}.  The grid is therefore not a design
choice: its alphabet is the metric-entropy count of distinguishable
$Z$-rotations at the working separation, and generic angles on it cost
$\Theta(\log(1/\epsilon))$ $T$ gates under Ross--Selinger
synthesis~\cite[Alg.~9.9]{ross2016optimal}.  No angle-by-angle $T$-count
equality is claimed; special grid points, such as Clifford angles, are
cheaper.
